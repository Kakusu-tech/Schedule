    const LED_COUNT = 44;

    const SLOPE_LEN = Math.sqrt(160 * 160 + 34 * 34);

    const HORIZ_LEN = 155;

    const TOTAL_LEN = SLOPE_LEN + HORIZ_LEN;


    function getLedCoordinates(t) {

      const dist = t * TOTAL_LEN;

      if (dist <= SLOPE_LEN) {

        const localT = dist / SLOPE_LEN;

        return {

          x: 20 + localT * (180 - 20),

          y: 52 + localT * (18 - 52)

        };

      } else {

        const localT = (dist - SLOPE_LEN) / HORIZ_LEN;

        return {

          x: 180 + localT * (335 - 180),

          y: 18

        };

      }

    }

          current.targetTime = deadline.getTime();

          current.durationMins = spanMins;

        } else {

          current.targetTime = Date.now() + (current.durationMins || 15) * 60000;

          current.createdAt = Date.now();

        }

        current.firedOffsets = current.firedOffsets || [];

        hasNotifiedCurrentTask = false;

        localStorage.setItem('booster_tasks', JSON.stringify(tasks));

      }


      if (!Array.isArray(current.notifyOffsets)) {

        current.notifyOffsets = getNotificationOffsetsForRemainingMs(current.targetTime - Date.now());

      }

      if (!Array.isArray(current.firedOffsets)) {

        current.firedOffsets = [];

      }

      if (current.deadlineTime) {

        document.getElementById('target-deadline-badge').innerText = `TARGET: 〜${current.deadlineTime}`;

      const update = () => {

        const diff = current.targetTime - Date.now();

        if (diff > 0) {

          const totalSpan = current.targetTime - current.createdAt;

          const remainingRatio = totalSpan > 0 ? Math.max(0, Math.min(1, diff / totalSpan)) : 0;

          

          const elapsedRatio = 1.0 - remainingRatio;

          updateLedMatrix(elapsedRatio);

          document.getElementById('progress-pct-label').innerText = `${Math.round(remainingRatio * 100)}%`;


          const unitLabel = document.getElementById('timer-unit-label');

          if (diff >= 3600000) {

            const hours = Math.floor(diff / 3600000);

            const mins = Math.floor((diff % 3600000) / 60000);

            document.getElementById('time-left').innerText = `${String(hours).padStart(2, '0')}:${String(mins).padStart(2, '0')}`;

            if (unitLabel) unitLabel.innerText = "HRS:MIN";

          } else {

            const min = Math.floor(diff / 60000);

            const sec = Math.floor((diff % 60000) / 1000);

            document.getElementById('time-left').innerText = `${String(min).padStart(2, '0')}:${String(sec).padStart(2, '0')}`;

            if (unitLabel) unitLabel.innerText = "MIN:SEC";

          }


          // 残り時間別の段階的通知（起動中・フォアグラウンド時）

          current.notifyOffsets.forEach((offsetMin) => {

            if (offsetMin <= 0 || current.firedOffsets.includes(offsetMin)) return;

            const thresholdMs = offsetMin * 60000;

            if (diff <= thresholdMs && diff > Math.max(0, thresholdMs - 60000)) {

              current.firedOffsets.push(offsetMin);

              localStorage.setItem('booster_tasks', JSON.stringify(tasks));

              playWarmTap();

              sendSystemNotification("FOCUS", `未完了タスクあり（残り${offsetMin}分）: ${current.title}`, {

                tag: `focus-task-${current.id}-${offsetMin}`,

                renotify: true,

                data: { url: location.href }

              });

            }

          });


          document.getElementById('cheer-message').innerText = current.type === 'main' ? "MANDATORY ROUTINE ENGAGED" : "CONCENTRATE ON CURRENT OPERATION";

        } else {

          updateLedMatrix(1.0);

          document.getElementById('progress-pct-label').innerText = "0%";

          document.getElementById('time-left').innerText = "00:00";

          document.getElementById('timer-unit-label').innerText = "MIN:SEC";

          document.getElementById('cheer-message').innerText = "TIME EXPIRED // SEQUENCE SUSPENDED";


          if (!hasNotifiedCurrentTask) {

            hasNotifiedCurrentTask = true;

            playWarmMajorBrass();

            const isZeroTarget = current.notifyOffsets.includes(0) && !current.firedOffsets.includes(0);

            if (isZeroTarget) {

              current.firedOffsets.push(0);

              localStorage.setItem('booster_tasks', JSON.stringify(tasks));

            }

            const bodyText = isZeroTarget

              ? `未完了タスクあり（残り時間0分0秒）: ${current.title}`

              : `目標時間が終了しました: ${current.title}`;


            sendSystemNotification("FOCUS", bodyText, {

              tag: `focus-task-${current.id}-0`,

              renotify: true,

              data: { url: location.href }

            });

          }

        }

      };

      update();

      timerInterval = setInterval(update, 1000);

    }

        current.firedOffsets = [];

        hasNotifiedCurrentTask = false;

        saveAndRefresh();

      }

    }
